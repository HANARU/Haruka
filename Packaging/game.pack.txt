# Haruka game pack manifest (2026-10-24). Read by HarukaPak (src/tools/HarukaPak.cpp); the build runs it for the
# GamePack target and the Windows package (cmake --build build --config Release --target PACKAGE) ships the result
# as game.hpak next to HarukaGame.exe. Design: Docs/Packaging_Design.md section 5.
#
# One command per line. ${NAME} comes from -D NAME=VALUE on the HarukaPak command line.
#   map <folder> <logical prefix>   every file under the folder, as "<prefix>/<relative path>"
#   exclude <pattern>...            logical paths to leave out
#   stored <pattern>...             keep as is (already compressed formats); everything else is LZ4
#   text <pattern>...               convert CRLF to LF before packing, so every checkout gives the same pack
#   textures <off|bc|astc> <cache> <model pattern>...   compress the textures of the packed models to KTX2 and add them
#                                   as textures/<target>/<key>.ktx2 (Docs/Texture_Compression.md D6). Results are kept in
#                                   <cache> and reused by the next pack.
# Patterns: * = any characters except '/', ** = any characters including '/', ? = one character.
# A pattern without '/' is matched against the file name only. Letter case is ignored.

map ${PROJECT_DIR}/Content /Game
map ${PROJECT_DIR}/Config /Config
map ${SHADER_DIR} shaders
# Engine content (2026-11-19, glyph system T4-4): the Pretendard fonts for UI panels and the localization
# tables. The game mounts it as /Engine/ like the editor does.
map ${ENGINE_CONTENT_DIR} /Engine

# Derived data and leftovers. The glTF cache is rebuilt on the device under the write folder.
# **/.* leaves out hidden files (name starts with '.'), **/.*/** everything under a hidden folder (.svn, .vs).
exclude /Game/cache/** **/*.tmp **/.* **/.*/** Thumbs.db desktop.ini
exclude shaders/*.d
# The .pot is the translation template for translators; the game only reads the .po tables. Editor icons stay
# in the editor (the game has no editor UI).
exclude /Engine/Localization/*.pot /Engine/Editor/**

# Local-only test assets on this machine (2026-11-06, also in .gitignore). Never ship them.
exclude /Game/models/_reference/** /Game/scenes/_reference/**

# Pre-compressed textures (2026-11-21): BC for the Windows pack, ASTC for the Android pack (TEXTURE_TARGET comes from
# cmake/HarukaProject.cmake). The original images stay in the models; the game uses them when the device lacks the format.
textures ${TEXTURE_TARGET} ${TEXTURE_CACHE} *.glb *.gltf

stored *.ogg *.mp3 *.flac *.opus *.wav *.png *.jpg *.jpeg *.ktx2 *.basis *.webp *.zip
text *.hscene *.hprefab *.ini *.txt *.po *.svg
